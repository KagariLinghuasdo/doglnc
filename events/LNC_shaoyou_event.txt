add_namespace = LNC_shaoyou_event

country_event = {
	id = LNC_shaoyou_event.1
	title = LNC_shaoyou_event.1.t
	desc = LNC_shaoyou_event.1.desc
	is_triggered_only = yes
	immediate = {
		#民用
		set_variable = { shaoyou_stability_weekly = 0.001 }
		set_variable = { shaoyou_opinion_gain_monthly_factor = 0.15 }
		set_variable = { shaoyou_production_speed_infrastructure_factor= 0.25 }
		set_variable = { shaoyou_production_speed_industrial_complex_factor= 0.15 }
		set_variable = { shaoyou_shaoyou_production_speed_arms_factory_factor = 0.15 }
		set_variable = { shaoyou_research_speed_factor = 0.10 }
		#军用
		set_variable = { shaoyou_industrial_capacity_factory = 0.10 }
		set_variable = { shaoyou_industrial_capacity_dockyard = 0.10 }
		set_variable = { shaoyou_production_factory_max_efficiency_factor = 0.25}
		set_variable = { shaoyou_production_factory_efficiency_gain_factor= 0.50 }
		#军队
		set_variable = { shaoyou_land_reinforce_rate = 0.02 }
		set_variable = { shaoyou_coordination_bonus = 0.02 }
		set_variable = { shaoyou_planning_speed  = 0.50 }
		set_variable = { shaoyou_max_planning = 0.50 }

	}

	option = {
		name = LNC_shaoyou_event.civilian
		add_political_power = 50
		add_offsite_building = {
			type = industrial_complex
			level = 3
    	}
		lnc_remove_ai_guided_construction = yes
		add_dynamic_modifier = {
			modifier = lnc_ai_guided_construction_industrial
			days = 85 
		}
	}
	option = {
		name = LNC_shaoyou_event.military
		add_offsite_building = {
			type = arms_factory
			level = 3
    	}
		lnc_remove_ai_guided_construction = yes
		add_dynamic_modifier = {
			modifier = lnc_ai_guided_construction_production
			days = 85 
		}
	}
	option = {
		name = LNC_shaoyou_event.army
		lnc_remove_ai_guided_construction = yes
		add_dynamic_modifier = {
			modifier = lnc_ai_guided_construction_army
			days = 85 
		}
	}
}
